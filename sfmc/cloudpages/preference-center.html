%%[ /* ========================================================================== NORTHWIND LEARNING | SFMC CLOUDPAGE PREFERENCE CENTRE Author: Faisal Irfan | github.com/faisal8143 -------------------------------------------------------------------------- - Identifies the subscriber securely from an encrypted email link (CloudPagesURL passes SubscriberKey, never a raw email in the URL) - Reads and saves topic preferences in a Data Extension - Global unsubscribe via LogUnsubEvent so All Subscribers stays in sync - Single page: GET shows the form, POST saves and shows a confirmation ========================================================================== */ SET @sk = AttributeValue("_subscriberkey") IF Empty(@sk) THEN SET @sk = RequestParameter("sk") ENDIF SET @valid = IIF(Empty(@sk), false, true) IF @valid AND RequestParameter("submitted") == "1" THEN SET @news = IIF(RequestParameter("news") == "on", "true", "false") SET @events = IIF(RequestParameter("events") == "on", "true", "false") SET @offers = IIF(RequestParameter("offers") == "on", "true", "false") SET @freq = RequestParameter("frequency") IF IndexOf("weekly|monthly", @freq) == 0 THEN SET @freq = "monthly" ENDIF UpsertData("Subscriber_Preferences", 1, "SubscriberKey", @sk, "News", @news, "Events", @events, "Offers", @offers, "Frequency", @freq, "UpdatedAt", Now()) IF RequestParameter("unsub_all") == "on" THEN SET @jid = RequestParameter("jid") SET @lid = RequestParameter("lid") SET @rr = CreateObject("ExecuteRequest") SetObjectProperty(@rr, "Name", "LogUnsubEvent") SET @p1 = CreateObject("APIProperty") SetObjectProperty(@p1, "Name", "SubscriberKey") SetObjectProperty(@p1, "Value", @sk) AddObjectArrayItem(@rr, "Parameters", @p1) SET @p2 = CreateObject("APIProperty") SetObjectProperty(@p2, "Name", "Reason") SetObjectProperty(@p2, "Value", "Preference centre: unsubscribe from all") AddObjectArrayItem(@rr, "Parameters", @p2) SET @status = InvokeExecute(@rr, @msg, @errCode) ENDIF SET @saved = true ENDIF IF @valid THEN SET @row = LookupRows("Subscriber_Preferences", "SubscriberKey", @sk) IF RowCount(@row) > 0 THEN SET @r = Row(@row, 1) SET @news = Field(@r, "News") SET @events = Field(@r, "Events") SET @offers = Field(@r, "Offers") SET @freq = Field(@r, "Frequency") ENDIF ENDIF ]%% Email preferences | Northwind Learning
