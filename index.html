sfmc/ampscript/personalization.amp
%%[
/*
  Personalisation block for the master template.
  - Pulls the subscriber's course enquiry from a Data Extension
  - Falls back to safe defaults so the email never renders empty values
  - Builds a tracked CTA URL with UTM parameters
*/

SET @subscriberKey = _subscriberkey
SET @firstName     = AttributeValue("FirstName")

IF Empty(@firstName) THEN
  SET @firstName = "there"
ELSE
  SET @firstName = ProperCase(Trim(@firstName))
ENDIF

/* Latest enquiry for this subscriber */
SET @rows = LookupOrderedRows("Course_Enquiries", 1, "EnquiryDate DESC", "SubscriberKey", @subscriberKey)

IF RowCount(@rows) > 0 THEN
  SET @row        = Row(@rows, 1)
  SET @courseName = Field(@row, "CourseName")
  SET @courseCode = Field(@row, "CourseCode")
ELSE
  SET @courseName = "your chosen course"
  SET @courseCode = "GENERAL"
ENDIF

/* Term details from a lookup DE, with a fallback */
SET @termName  = Lookup("Term_Config", "TermName",  "IsCurrent", "true")
SET @closeRaw  = Lookup("Term_Config", "CloseDate", "IsCurrent", "true")

IF Empty(@termName) THEN SET @termName = "the upcoming term" ENDIF

IF NOT Empty(@closeRaw) THEN
  SET @closeDate = FormatDate(@closeRaw, "d MMMM yyyy", "", "en-AU")
ELSE
  SET @closeDate = "soon"
ENDIF

/* Tracked CTA */
SET @baseUrl = "https://www.example.edu/enrol"
SET @ctaUrl  = Concat(
  @baseUrl,
  "?course=", URLEncode(@courseCode, 1, 1),
  "&utm_source=sfmc&utm_medium=email&utm_campaign=", URLEncode(emailname_, 1, 1)
)

/* Dynamic content: different message for returning students */
SET @isReturning = Lookup("Student_Status", "IsReturning", "SubscriberKey", @subscriberKey)
]%%

  • Upsert keyed on SubscriberKey
  • Errors logged to a DE, never shown to visitors
  • JSON response for a clean front-end UX
sfmc/ssjs/cloudpage-form-handler.ssjs
<script runat="server">
  /*
    CloudPage form handler
    - Accepts a POST from a preference / enquiry form
    - Validates and sanitises input
    - Upserts into a Data Extension keyed on SubscriberKey
    - Returns JSON so the front end can show success/error states
  */
  Platform.Load("core", "1.1.1");

  var response = { ok: false, message: "" };

  try {
    if (Request.Method() != "POST") {
      response.message = "Invalid request method.";
    } else {
      var email     = Request.GetFormField("email");
      var firstName = Request.GetFormField("firstName");
      var course    = Request.GetFormField("courseCode");
      var optIn     = Request.GetFormField("optIn") == "on";

      var emailOk = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email || "");

      if (!emailOk) {
        response.message = "Please enter a valid email address.";
      } else if (!course) {
        response.message = "Please choose a course.";
      } else {
        var key = String(email).toLowerCase();

        var de = DataExtension.Init("Course_Enquiries");
        de.Rows.Upsert(
          { SubscriberKey: key },
          ["EmailAddress", "FirstName", "CourseCode", "OptIn", "EnquiryDate"],
          [key, String(firstName || "").substring(0, 50), course, optIn, Platform.Function.Now()]
        );

        response.ok = true;
        response.message = "Thanks! We will be in touch shortly.";
      }
    }
  } catch (e) {
    /* Log the error to a DE instead of exposing details to the visitor */
    try {
      DataExtension.Init("CloudPage_Error_Log").Rows.Add({
        Page: "enquiry-handler",
        Error: Stringify(e).substring(0, 4000),
        LoggedAt: Platform.Function.Now()
      });
    } catch (ignore) {}
    response.message = "Something went wrong. Please try again.";
  }

  Platform.Response.SetResponseHeader("Content-Type", "application/json");
  Write(Stringify(response));
</script>

  • 90-day opens and clicks from _Open and _Click
  • Weighted score with Hot / Warm / Cold segments
  • Excludes hard bounces and inactive subscribers
  • Overwrite target DE for daily refresh
sfmc/sql/engaged-subscribers-90d.sql
/*
  Automation Studio query: engagement score for the last 90 days
  Target DE: Engagement_Score_90d (Overwrite)
  Primary key: SubscriberKey

  Score = opens x 1 + clicks x 3, excluding bounced/unsubscribed subscribers.
  Uses system data views (_Open, _Click, _Bounce, _Subscribers).
*/

SELECT
    s.SubscriberKey,
    s.EmailAddress,
    ISNULL(o.Opens, 0)                                   AS Opens90d,
    ISNULL(c.Clicks, 0)                                  AS Clicks90d,
    ISNULL(o.Opens, 0) * 1 + ISNULL(c.Clicks, 0) * 3     AS EngagementScore,
    CASE
        WHEN ISNULL(c.Clicks, 0) >= 3                    THEN 'Hot'
        WHEN ISNULL(o.Opens, 0) >= 2 OR c.Clicks >= 1    THEN 'Warm'
        ELSE 'Cold'
    END                                                  AS Segment,
    GETDATE()                                            AS CalculatedAt
FROM _Subscribers s
LEFT JOIN (
    SELECT SubscriberKey, COUNT(DISTINCT JobID) AS Opens
    FROM _Open
    WHERE EventDate >= DATEADD(DAY, -90, GETDATE())
      AND IsUnique = 1
    GROUP BY SubscriberKey
) o ON o.SubscriberKey = s.SubscriberKey
LEFT JOIN (
    SELECT SubscriberKey, COUNT(DISTINCT JobID) AS Clicks
    FROM _Click
    WHERE EventDate >= DATEADD(DAY, -90, GETDATE())
      AND IsUnique = 1
    GROUP BY SubscriberKey
) c ON c.SubscriberKey = s.SubscriberKey
WHERE s.Status = 'active'
  AND NOT EXISTS (
      SELECT 1
      FROM _Bounce b
      WHERE b.SubscriberKey = s.SubscriberKey
        AND b.BounceCategory = 'Hard bounce'
        AND b.EventDate >= DATEADD(DAY, -90, GETDATE())
  )
