    ========================================================================
    CloudPage form handler
    - Accepts a POST from a preference / enquiry form

/*
  ==========================================================================
  NORTHWIND LEARNING | ENGAGEMENT SCORING QUERY
  Author: Faisal Irfan | github.com/faisal8143
  ==========================================================================
  Automation Studio query: engagement score for the last 90 days
  Target DE: Engagement_Score_90d (Overwrite)
  Primary key: SubscriberKey

  Score = opens x 1 + clicks x 3, excluding bounced/unsubscribed subscribers.
  Uses system data views (_Open, _Click, _Bounce, _Subscribers).
*/

SELECT
    s.SubscriberKey,
    s.EmailAddress,
    ISNULL(o.Opens, 0)                                   AS Opens90d,
    ISNULL(c.Clicks, 0)                                  AS Clicks90d,
    ISNULL(o.Opens, 0) * 1 + ISNULL(c.Clicks, 0) * 3     AS EngagementScore,
    CASE
        WHEN ISNULL(c.Clicks, 0) >= 3                    THEN 'Hot'
        WHEN ISNULL(o.Opens, 0) >= 2 OR c.Clicks >= 1    THEN 'Warm'
        ELSE 'Cold'
    END                                                  AS Segment,
    GETDATE()                                            AS CalculatedAt
FROM _Subscribers s
LEFT JOIN (
    SELECT SubscriberKey, COUNT(DISTINCT JobID) AS Opens
    FROM _Open
    WHERE EventDate >= DATEADD(DAY, -90, GETDATE())
      AND IsUnique = 1
    GROUP BY SubscriberKey
) o ON o.SubscriberKey = s.SubscriberKey
LEFT JOIN (
    SELECT SubscriberKey, COUNT(DISTINCT JobID) AS Clicks
    FROM _Click
    WHERE EventDate >= DATEADD(DAY, -90, GETDATE())
      AND IsUnique = 1
    GROUP BY SubscriberKey
) c ON c.SubscriberKey = s.SubscriberKey
WHERE s.Status = 'active'
  AND NOT EXISTS (

<script runat="server">
  /*
    ========================================================================
    NORTHWIND LEARNING | SSJS WSPROXY TRIGGERED SEND
    Author: Faisal Irfan | github.com/faisal8143
    ------------------------------------------------------------------------
    Sends a transactional "enquiry received" email from a CloudPage using
    WSProxy (faster and lighter than the Core library for SOAP calls).
      - Looks up the Triggered Send Definition by external key
      - Passes personalisation attributes
      - Retries once on a transient error and logs the outcome
    ========================================================================
  */
  Platform.Load("core", "1.1.1");
  var api = new Script.Util.WSProxy();

  function sendEnquiryEmail(email, firstName, courseName) {
    var payload = {
      TriggeredSendDefinition: { CustomerKey: "TSD_Enquiry_Received" },
      Subscribers: [{
        EmailAddress: email,
        SubscriberKey: email.toLowerCase(),
        Attributes: [
          { Name: "FirstName",  Value: firstName || "there" },
          { Name: "CourseName", Value: courseName || "your chosen course" }
        ]
      }]
    };
    var res = api.createItem("TriggeredSend", payload);
    if (res.Status !== "OK") {           /* one retry for transient errors */
      res = api.createItem("TriggeredSend", payload);
    }
    return res;
  }

  try {
    var r = sendEnquiryEmail(
      Request.GetFormField("email"),
      Request.GetFormField("firstName"),
      Request.GetFormField("courseName")
    );
    DataExtension.Init("Send_Log").Rows.Add({
      SubscriberKey: String(Request.GetFormField("email")).toLowerCase(),
      Status: r.Status,
      Message: String(r.Results && r.Results[0] ? r.Results[0].StatusMessage : "").substring(0, 500),
      LoggedAt: Platform.Function.Now()
    });
  } catch (e) {
    Write("<!-- send failed -->");
  }

/*
  ==========================================================================
  NORTHWIND LEARNING | DEDUPLICATE ENQUIRIES (LATEST PER SUBSCRIBER)
  Author: Faisal Irfan | github.com/faisal8143
  ==========================================================================
  Automation Studio query. Source: Course_Enquiries_Raw (all form fills)
  Target: Course_Enquiries (Overwrite, PK SubscriberKey)

  Keeps only each subscriber's most recent enquiry, normalises email case,
  and drops rows without a valid-looking email address.
*/

SELECT
    LOWER(LTRIM(RTRIM(e.EmailAddress)))  AS SubscriberKey,
    LOWER(LTRIM(RTRIM(e.EmailAddress)))  AS EmailAddress,
    e.FirstName,
    e.CourseCode,
    e.CourseName,
    e.OptIn,
    e.EnquiryDate
FROM (
    SELECT
        r.*,
        ROW_NUMBER() OVER (
            PARTITION BY LOWER(LTRIM(RTRIM(r.EmailAddress)))
            ORDER BY r.EnquiryDate DESC
        ) AS rn
    FROM Course_Enquiries_Raw r
    WHERE r.EmailAddress LIKE '%_@_%._%'
) e
WHERE e.rn = 1

/*
  ==========================================================================
  NORTHWIND LEARNING | CUSTOM FORM -> PARDOT FORM HANDLER
  Author: Faisal Irfan | github.com/faisal8143
  --------------------------------------------------------------------------
  Posts a custom website form to a Pardot (Account Engagement) form
  handler without leaving the page, while keeping Pardot's visitor
  tracking (cookies) and completion actions working.
    - Client-side validation with accessible error messages
    - Submits via a hidden iframe (form handlers do not support CORS)
    - UTM and GCLID values captured into hidden fields for attribution
    - Prevents double submits; shows a success state
  ==========================================================================
*/
(function () {
  var HANDLER_URL = "https://go.example.edu/l/000000/2026-10-30/abcde"; /* form handler endpoint */
  var form = document.getElementById("enquiry-form");
  if (!form) return;

  /* 1. Capture attribution parameters into hidden fields */
  var params = new URLSearchParams(window.location.search);
  ["utm_source", "utm_medium", "utm_campaign", "utm_content", "gclid"].forEach(function (key) {
    var field = form.querySelector('[name="' + key + '"]');
    if (field && params.get(key)) field.value = params.get(key);
  });

  /* 2. Hidden iframe target keeps the visitor on the page */
  var frame = document.createElement("iframe");
  frame.name = "pardot-target";
  frame.title = "Form submission";
  frame.hidden = true;
  document.body.appendChild(frame);

  function setError(input, message) {
    var id = input.id + "-error";
    var el = document.getElementById(id);
    if (!el) {
      el = document.createElement("p");
      el.id = id;
      el.className = "field-error";
      input.insertAdjacentElement("afterend", el);
    }
    el.textContent = message;
    input.setAttribute("aria-invalid", message ? "true" : "false");
    input.setAttribute("aria-describedby", id);
  }

  function validate() {
    var ok = true;
    var email = form.elements.email;
    var name = form.elements.first_name;
    setError(name, name.value.trim() ? "" : "Please enter your first name.");
    setError(email, /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.value) ? "" : "Please enter a valid email address.");
    if (!name.value.trim() || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.value)) ok = false;
    return ok;
  }

  form.addEventListener("submit", function (e) {
    if (!validate()) { e.preventDefault(); return; }
    var btn = form.querySelector('[type="submit"]');
    btn.disabled = true;
    btn.textContent = "Sending...";
    form.action = HANDLER_URL;
    form.method = "post";
    form.target = "pardot-target";
    frame.addEventListener("load", function () {
      form.hidden = true;
      var done = document.getElementById("enquiry-success");
      if (done) { done.hidden = false; done.focus(); }
    }, { once: true });
  });
})();
